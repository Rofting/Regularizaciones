# ============================================================
# 4. FLUJO NORMAL — PROCESAR UNA FACTURA NUEVA
# ============================================================

# Cuando llegue una factura de gas (ej: basergas644.pdf):
# 1. Ponla en la carpeta GestionFincas/entrada/
# 2. Ejecuta:

python -c "
import sys; sys.path.insert(0, '.')
from lector_pdf import procesar_archivo
from gestor_bd import conectar, buscar_comunidad_por_cif, insertar_factura, marcar_archivo_procesado, archivo_ya_procesado
import shutil, os

CARPETA_ENTRADA   = '../entrada'
CARPETA_PROCESADOS = '../procesados'
RUTA_BD = '../data/gestion.db'

con = conectar(RUTA_BD)

archivos = [f for f in os.listdir(CARPETA_ENTRADA) if f.lower().endswith('.pdf')]
for nombre in archivos:
    ruta = os.path.join(CARPETA_ENTRADA, nombre)

    # ¿Ya fue procesado?
    if archivo_ya_procesado(con, nombre):
        print(f'⏭️  Ya procesado: {nombre}')
        continue

    # Detectar comunidad por nombre del archivo
    # Convención: el archivo empieza con el código de comunidad
    # ej: '644_basergas.pdf' -> código '644'
    codigo = nombre.split('_')[0] if '_' in nombre else '644'

    resultado = procesar_archivo(ruta, codigo, con_bd=con)

    if resultado['ok'] and resultado['tipo'] == 'FACTURA':
        datos = resultado['datos']
        # Obtener id_comunidad e id_periodo
        com = buscar_comunidad_por_cif(con, resultado['cif_pdf'])
        if not com:
            print(f'⚠️  CIF {resultado[\"cif_pdf\"]} no encontrado en BD: {nombre}')
            continue
        # Obtener o crear periodo (por defecto el más reciente abierto)
        per = con.execute(
            \"SELECT id_periodo FROM periodos WHERE id_comunidad=? AND estado='abierto' ORDER BY fecha_inicio DESC LIMIT 1\",
            (com['id_comunidad'],)
        ).fetchone()
        if not per:
            print(f'⚠️  Sin periodo abierto para comunidad {codigo}')
            continue
        datos['id_comunidad'] = com['id_comunidad']
        datos['id_periodo']   = per['id_periodo']
        id_fac = insertar_factura(con, datos)
        if id_fac:
            marcar_archivo_procesado(con, nombre, resultado['hash_md5'], id_fac)
            shutil.move(ruta, os.path.join(CARPETA_PROCESADOS, nombre))
            print(f'✅ {nombre} → {datos[\"tipo_suministro\"]} {datos[\"importe_total\"]}€')
        else:
            print(f'⚠️  Duplicado (ya estaba en BD): {nombre}')
    elif resultado['ok'] and resultado['tipo'] == 'LECTURA_METRIGEST':
        print(f'📊 Metrigest: {len(resultado[\"datos\"][\"vecinos\"])} lecturas — pendiente de importar a BD')
    else:
        print(f'❌ {nombre}: {resultado[\"motivo\"]} — {resultado.get(\"detalle\",\"\")}')

con.close()
"
